# 1. Ensure Ubuntu's Bluetooth & D-Bus broker are active
sudo apt update && sudo apt install -y bluez dbus-broker
sudo systemctl enable --now dbus-broker bluetooth

# 2. Create the directory and docker-compose.yml file
mkdir -p ~/docker/homeassistant/config && cd ~/docker/homeassistant

cat << 'EOF' > docker-compose.yml
services:
  homeassistant:
    container_name: homeassistant
    image: ghcr.io/home-assistant/home-assistant:stable
    restart: unless-stopped
    privileged: true
    network_mode: host
    environment:
      - TZ=Europe/Berlin
    volumes:
      - ./config:/config
      - /etc/localtime:/etc/localtime:ro
      - /run/dbus:/run/dbus:ro
EOF

# 3. Start Home Assistant
docker compose up -d

#for firewall:
sudo ufw allow 8123/tcp
